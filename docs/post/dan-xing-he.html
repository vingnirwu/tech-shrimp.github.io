        * {
            margin: 0;
            padding: 0;
            box-sizing: border-box;
        }

        .box {
            width: 600px;
            height: 800px;
            background-color: rgb(0, 110, 255);

            /* 设置弹性盒子 */
            /* 弹性盒子的子元素按主轴排列 */
            display: flex;
            /* 容器（父盒子）设置 display: flex; 可以让子盒子按照主轴方式排列。
            1.如果子元素有大小，则按照给定大小来显示。
            2.如果子元素没有大小，则拉伸充满父容器。
3           3.若子元素总宽度超过容器宽度，默认会压缩子元素。 
            4.如果子盒子没有设置宽度，默认会拉伸充满父容器。*/
            /* 注意：
            1.父控制子
            2.主轴与交叉轴 */
        }

        /* 主轴与交叉轴对齐方式 */
        .dq {
            /* 主轴对齐方式 */
            /* 有以下写法：
                1. flex-start：左对齐 ABC----
                2. flex-end：右对齐   ----ABC
                3. center：居中对齐   --ABC--
                4. space-between：项目之间等间距对齐    A--B--C--D
                5. space-around：项目之间等间距对齐，项目与边框距离相等  -A--B--C-
                6. space-evenly：项目等间距对齐，项目与边框距离相等 --A--B--C--*/
            justify-content: space-between;
            /* 交叉轴对齐方式 */
            /* 有以下写法：
                1. flex-start：上对齐
                2. flex-end：下对齐
                3. center：居中对齐
                4. stretch：拉伸对齐 注意：子盒子没有设置高度时，才会拉伸充满父容器
                5. baseline：基线对齐 */
            align-items: stretch;
        }

        /* 改变主轴方向 */
        .diraction {
            display: flex;
            /* 属性值：
                1. row：主轴方向为水平方向
                2. row-reverse：主轴方向为水平方向，从右到左
                3. column：主轴方向为垂直方向
                4. column-reverse：主轴方向为垂直方向，从下到上 */
            flex-direction: column-reverse;
            /* 注意：必须给父元素设置 */
            /* 如果此时给ul，不会改变排列方式 */
            /* 对齐也需要注意：主轴改变了，交叉轴也改变了 */
            justify-content: center;
            align-items: center;

            margin-right: 20px;
        }

        .sanlian li {
            display: flex;
            border: 1px solid red;
            width: 300px;
            height: 300px;
            font-size: 25px;
            flex-direction: column;
        }


        /* 是否换行 */
        .linebreak {
            height: 400px;
            align-items: flex-start;
            justify-content: flex-start;

            /* 换行 */
            /* 属性值：
                1. wrap：换行
                2. nowrap：不换行 
                3. wrap-reverse：换行，从下到上
                4. nowrap-reverse：不换行，从右到左
                默认为不换行*/
            flex-wrap: wrap;

            /* 多行的对齐 */
            /* align-content: 对齐方式; */
            /* 属性值：
                1. flex-start：上对齐
                2. flex-end：下对齐
                3. center：居中对齐
                4. space-between：项目之间等间距对齐    A--B--C--D
                5. space-around：项目之间等间距对齐，项目与边框距离相等  -A--B--C-
                6. space-evenly：项目等间距对齐，项目与边框距离相等 --A--B--C-- */
            align-content: flex-start;
        }

        /* 子项目flex伸缩与gap间隙 */
        /* flex */
        .box2 {
            display: flex;
            flex-wrap: wrap;
            width: 80%;
            border: 5px solid black;
            margin: 0 auto;
        }

        .box2 .item {
            width: 100px;
            height: 300px;
            /* margin-left: 5px; */

            /* flex伸缩 */
            /* flex写在小盒子上，优先级比width、height高，会覆盖width、height */
            /* 1 表示每个盒子占1份空间 */
            flex: 1;

            /* flex: 1;是个简写  是flex-grow: 1;  flex-shrink: 1;  flex-basis: 0% 的简写 */
            /* flex-grow: 2; 放大比例，放大时他等这个比例放大
            flex-shrink: 1; 缩小比例，缩小时他等这个比例缩小
            flex-basis: 0%; 基础宽度，默认是auto，会覆盖width、height */

            background-color: pink;
            border-radius: 10px;
        }

        /* 选择类名为 box的元素下，第 n 个子元素，并且这个子元素必须同时满足 .item的类名选择器条件。 */
        .box2 .item:nth-child(2) {
            flex: 2;
        }

        /* gap */
        .box2 {
            /* gap 表示子元素之间的间距 */
            /* 可以设置水平间距和垂直间距 */
            /* gap：10px 表示水平间距和垂直间距都是10px */
            /* gap：10px 20px 表示水平间距是10px，垂直间距是20px */
            gap: 20px 20px;
        }


        /* flex多行伸缩布局 */
        /* 具体看多行伸缩布局.html */
        .box .item {
            /* 用百分比来分列 */
            flex: 0 0 16.66666666%;
            height: 120px;
            background-color: rgb(255, 192, 225);
            margin-bottom: 8px;
            padding: 0 8px;
        }

        .item {
            width: 100px;
            height: 100px;
            background-color: pink;
            border-radius: 10px;
        }

        /* 拓展：图片的垂直居中 */
        /* 原本图片会对文字的下基线对齐，加上后与文字垂直居中 */
        .img1 {
            width: 230px;
            height: 130px;
            vertical-align: middle;
        }
