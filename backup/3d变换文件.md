<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Document</title>
    <style>
        img {
            width: 200px;
            margin: 20px auto;
            transition: all .8s;
        }

        /* 旋转 */
        /* rotateX 旋转X轴 */
        /* rotateY 旋转Y轴 */
        /* rotateZ 旋转Z轴 */
        .xturn:hover {
            transform: rotateX(40deg);
        }

        .yturn:hover {
            transform: rotateY(140deg);
        }

        /* .zturn:hover {
            transform: rotateZ(180deg);
        } */

        /* 透视 */
        /* 父元素与子元素设置有区别 */
        /* 父元素设置：
        perspective: 5000px;
        数值越小，透视效果越明显
        子元素设置：
        transform: perspective(500px) rotateX(40deg); 
        子元素中perspective必须是transform的第一个参数，否则不会生效*/


        /* 给父元素设置透视 */
        body {
            perspective: 200px;
        }

        /* 给子元素设置透视 */
        .zturn:hover {
            transform: perspective(200px) rotateX(40deg) rotateZ(40deg);
        }

        /* 3d位移 */
        /* transform: translate3d(x, y, z); */
        /* 主要： transform: translateZ(z); */
        .remove:hover {
            transform: translateZ(60px);
        }

        /* 子元素需要找到父元素，开启3d效果
        transform-style: preserve-3d; */
        .box {
            position: relative;
            width: 300px;
            height: 450px;
            margin: 0 auto;
            background-color: #a63434;

            /* 添加透视效果 */
            perspective: 1000px;
        }

        .box>div {
            position: absolute;
            top: 0;
            left: 0;
            width: 100%;
            height: 100%;
            background-color: pink;
            text-align: center;
            line-height: 450px;
            color: #fff;
            font-size: 30px;
            transition: all 1s;

            /* 隐藏反面的背面 */
            backface-visibility: hidden;
            /* 需要找到父元素，开启3d效果 */
            transform-style: preserve-3d;
        }


        .box .back {
            background-color: #176494;
            transform: rotateY(180deg);
        }

        .box .front {
            z-index: 1;
        }

        .box:hover .front {
            transform: rotateY(-180deg);
        }

        .box:hover .back {
            transform: rotateY(0deg);
        }

        /* 给字体添加3d效果 */
        .box .front p {
            font-size: 30px;
            line-height: 240px;
            /* 立体 */
            transform: translate3d(0, 0, 60px);
            backface-visibility: hidden;
        }

        .box .back p {
            font-size: 30px;
            line-height: 120px;
            transform: translate3d(0, 0, 60px);
            backface-visibility: hidden;
        }
    </style>
</head>
