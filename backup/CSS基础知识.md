```"html"
<!DOCTYPE html>
<html lang="en">

<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Document</title>

    css一般写在head与/head之间
    <style>
        /* 格式：选择器{样式} */
        p {
            color: red;
            /* 颜色  */
            font-size: 12px;
            /* 文字大小：12像素 */
        }
    </style>

    规范：
    1、属性与属性值对应
    2、属性与属性值之间用冒号:分割，用分号;结尾
</head>

<body>
    <p>段落标签</p>
    <p>文本1</p>
</body>

</html>
```
